#include <iostream>
#include <algorithm>    // sort, max_element, min_element
using namespace std;

int main() {
    // ── Declare & Initialize ──────────────────────────
    int arr1[5];                      // uninitialized
    int arr2[] = {10, 20, 30, 40, 50};  // initialized, size auto
    int arr3[5] = {1, 2};              // partial: arr3[2..4] = 0

    // ── Access ───────────────────────────────────────
    cout << arr2[0];      // 10  — first element
    cout << arr2[4];      // 50  — last element
    arr2[0] = 99;         // modify

    // ── Loop (index style) ───────────────────────────
    int len = sizeof(arr2) / sizeof(arr2[0]);
    for (int i = 0; i < len; i++)
        cout << arr2[i] << " ";

    // ── Range-based for ──────────────────────────────
    for (int n : arr2) cout << n << " ";

    // ── Algorithm functions ──────────────────────────
    sort(arr2, arr2 + 5);                       // ascending
    int mx = *max_element(arr2, arr2 + 5);      // max value
    int mn = *min_element(arr2, arr2 + 5);      // min value

    // ── Dynamic (heap) array ─────────────────────────
    int* dyn = new int[5];  // allocate at run-time
    delete[] dyn;             // MUST free!

    return 0;
}

#include <iostream>
using namespace std;

// ── Step 1: Node structure ──────────────────────────
struct Node {
    int   data;    // the value
    Node* next;    // pointer to the next node
};

// ── Step 2: Insert at END ───────────────────────────
void insertEnd(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;  n->next = NULL;
    if (head == NULL) { head = n; return; }
    Node* t = head;
    while (t->next != NULL) t = t->next;  // traverse to end
    t->next = n;
}

// ── Step 3: Insert at BEGIN ─────────────────────────
void insertBegin(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;
    n->next = head;   // point new node to old head
    head = n;         // update head — O(1)!
}

// ── Step 4: Delete at BEGIN ─────────────────────────
void deleteBegin(Node*& head) {
    if (head == NULL) return;
    Node* temp = head;
    head = head->next;  // advance head
    delete temp;         // free old head
}

// ── Step 5: Delete at END ──────────────────────────
void deleteEnd(Node*& head) {
    if (!head || !head->next) { delete head; head = NULL; return; }
    Node* prev = NULL, *curr = head;
    while (curr->next != NULL) { prev = curr; curr = curr->next; }
    prev->next = NULL;  delete curr;
}

// ── Step 6: Display ────────────────────────────────
void display(Node* head) {
    Node* t = head;
    while (t != NULL) {
        cout << t->data << " → ";
        t = t->next;
    }
    cout << "NULL\n";
}

int main() {
    Node* head = NULL;        // empty list
    insertEnd(head, 10);      // HEAD → 10 → NULL
    insertEnd(head, 20);      // HEAD → 10 → 20 → NULL
    insertEnd(head, 30);      // HEAD → 10 → 20 → 30 → NULL
    display(head);
    deleteBegin(head);        // HEAD → 20 → 30 → NULL
    display(head);
    return 0;
}
Delete at middle: Keep a prev pointer. When you reach the target node, do prev->next = current->next; delete current;. This skips and frees the node in O(n) (traverse) but O(1) relink.
📚 STRUCTURE 3

Stack

📚
Stack
LIFO — Last In, First Out

What is it? A Stack follows the LIFO (Last In, First Out) principle — the last element added is the first one removed. Think of a stack of plates: you always add and remove from the top.

After push(10), push(20), push(30)

TOP →
30 ← TOP
20
10

Operations:

  • push(val) — add to top
  • pop() — remove from top
  • peek() — view top without removing
  • isEmpty() — check if empty
push
O(1)
pop
O(1)
peek
O(1)
Search
O(n)

Real uses: Function call stack, undo/redo, expression evaluation, backtracking algorithms.

#include <iostream>
using namespace std;

const int MAX = 100;

struct Stack {
    int data[MAX];
    int top = -1;    // -1 means empty

    // Push — add element to top
    void push(int val) {
        if (top == MAX - 1) { cout << "Stack Overflow!\n"; return; }
        data[++top] = val;    // increment top, then store
    }

    // Pop — remove and return top element
    int pop() {
        if (top == -1) { cout << "Stack Underflow!\n"; return -1; }
        return data[top--];   // return top, then decrement
    }

    // Peek — view top without removing
    int peek()    { return top == -1 ? -1 : data[top]; }
    bool isEmpty(){ return top == -1; }
};

int main() {
    Stack s;
    s.push(10);   // stack: [10]
    s.push(20);   // stack: [10, 20]
    s.push(30);   // stack: [10, 20, 30]

    cout << "Top: "  << s.peek() << "\n";   // 30
    cout << "Pop: "  << s.pop()  << "\n";   // 30
    cout << "Top: "  << s.peek() << "\n";   // 20
    return 0;
}
🚶 STRUCTURE 4

Queue

🚶
Queue
FIFO — First In, First Out

What is it? A Queue follows the FIFO (First In, First Out) principle — the first element added is the first one removed. Think of a checkout line: new people join at the back, and are served from the front.

After enqueue(10), enqueue(20), enqueue(30):

FRONT →
10
20
30
← BACK
enqueue (add)
O(1)
dequeue (remove)
O(1)
front (peek)
O(1)
Search
O(n)

Real uses: Print queues, CPU scheduling, BFS graph traversal, keyboard input buffer.

#include <iostream>
using namespace std;

const int MAX = 100;

struct Queue {
    int  data[MAX];
    int  front = 0;
    int  back  = -1;
    int  size  = 0;

    // Enqueue — add to back
    void enqueue(int val) {
        if (size == MAX) { cout << "Queue Full!\n"; return; }
        back = (back + 1) % MAX;   // circular wrap
        data[back] = val;
        size++;
    }

    // Dequeue — remove from front
    int dequeue() {
        if (size == 0) { cout << "Queue Empty!\n"; return -1; }
        int val = data[front];
        front = (front + 1) % MAX;  // circular wrap
        size--;
        return val;
    }

    int  peek()    { return size == 0 ? -1 : data[front]; }
    bool isEmpty(){ return size == 0; }
};

int main() {
    Queue q;
    q.enqueue(10);   // queue: [10]
    q.enqueue(20);   // queue: [10, 20]
    q.enqueue(30);   // queue: [10, 20, 30]

    cout << "Front:   " << q.peek()     << "\n";   // 10
    cout << "Dequeue: " << q.dequeue() << "\n";   // 10
    cout << "Front:   " << q.peek()     << "\n";   // 20
    return 0;
}
Stack vs Queue: Both are linear data structures with O(1) add/remove. The key difference is direction — Stack removes from the same end it adds (LIFO), Queue removes from the opposite end (FIFO).
🌳 STRUCTURE 5

Binary Tree

🌳
Binary Tree / BST
Hierarchical · Each node has at most 2 children

What is it? A tree is a hierarchical (non-linear) data structure. Each element is a node. There is one root at the top. Each node has at most two children: left and right. A node with no children is called a leaf.

Binary Search Tree (BST) rule: left child < parent < right child — this makes searching very fast.

Search (BST)
O(log n)
Insert (BST)
O(log n)
Traversal
O(n)
Worst case
O(n)

Real uses: File systems, expression parsing, database indexes, autocomplete, game AI (decision trees).

Traversals

TraversalOrderOutput on tree above
InorderLeft → Root → Right20 30 40 50 60 70 80 (sorted!)
PreorderRoot → Left → Right50 30 20 40 70 60 80
PostorderLeft → Right → Root20 40 30 60 80 70 50
#include <iostream>
using namespace std;

// ── Node structure ─────────────────────────────────
struct TreeNode {
    int       data;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int val) : data(val), left(NULL), right(NULL) {}
};

// ── Insert into BST ────────────────────────────────
TreeNode* insert(TreeNode* root, int val) {
    if (root == NULL) return new TreeNode(val);
    if      (val < root->data) root->left  = insert(root->left,  val);
    else if (val > root->data) root->right = insert(root->right, val);
    return root;   // duplicates are ignored
}

// ── Inorder: Left → Root → Right (gives SORTED order) ─
void inorder(TreeNode* root) {
    if (root == NULL) return;
    inorder(root->left);
    cout << root->data << " ";
    inorder(root->right);
}

// ── Preorder: Root → Left → Right ─────────────────
void preorder(TreeNode* root) {
    if (root == NULL) return;
    cout << root->data << " ";
    preorder(root->left);
    preorder(root->right);
}

// ── Postorder: Left → Right → Root ────────────────
void postorder(TreeNode* root) {
    if (root == NULL) return;
    postorder(root->left);
    postorder(root->right);
    cout << root->data << " ";
}

// ── BST Search ─────────────────────────────────────
bool search(TreeNode* root, int key) {
    if (root == NULL)       return false;
    if (root->data == key)   return true;
    if (key < root->data)    return search(root->left,  key);
    else                     return search(root->right, key);
}

int main() {
    TreeNode* root = NULL;
    for (int v : {50, 30, 70, 20, 40, 60, 80})
        root = insert(root, v);

    cout << "Inorder  : "; inorder  (root); cout << "\n";
    cout << "Preorder : "; preorder (root); cout << "\n";
    cout << "Postorder: "; postorder(root); cout << "\n";

    cout << "Search 40: " << (search(root, 40) ? "Found" : "Not found") << "\n";
    cout << "Search 99: " << (search(root, 99) ? "Found" : "Not found") << "\n";
    return 0;
}

How to choose a structure:

  • Need fast index access? → Array
  • Lots of insertions/deletions? → Linked List
  • Undo / function calls / backtracking? → Stack
  • Scheduling / print queue / BFS? → Queue
  • Need sorted data + fast search? → Binary Search Tree

Q12 — Reverse an Array In-Place

Problem: Reverse an array in-place. Time O(n). Space O(1). No extra array allowed.

void reverseArray(int arr[], int n) {
    int left = 0, right = n - 1;
    while (left < right) {
        int temp   = arr[left];   // 3-step swap
        arr[left]  = arr[right];
        arr[right] = temp;
        left++;   right--;          // move pointers inward
    }
}
// Trace on [1,2,3,4,5]:
// left=0,right=4: swap 1↔5 → [5,2,3,4,1]
// left=1,right=3: swap 2↔4 → [5,4,3,2,1]
// left=2,right=2: stop (pointers met)
// Result: [5,4,3,2,1] ✓  Space O(1): only 3 variables

Q13 — Rotate Array Left by k Positions

Problem: Rotate an array left by k positions. Example: [1,2,3,4,5] rotated left by 2 → [3,4,5,1,2]. Time O(n). Space O(1).

void rotateLeft(int arr[], int n, int k) {
    k = k % n;                    // handle k ≥ n
    reverseArray(arr, k);         // Step 1: reverse first k elements
    reverseArray(arr + k, n - k); // Step 2: reverse remaining n-k
    reverseArray(arr, n);         // Step 3: reverse entire array
}
// 3-Reversal trick trace on [1,2,3,4,5], k=2:
// Step 1: reverse [1,2]    → [2,1,3,4,5]
// Step 2: reverse [3,4,5] → [2,1,5,4,3]
// Step 3: reverse all      → [3,4,5,1,2] ✓
Why 3 reversals work: Reversing first k, then last n-k, then the whole array achieves rotation in 3 × O(n/2) = O(n) time with zero extra memory. This is called the reversal algorithm.

Q14 — Find the Second Largest Element (Single Pass)

Problem: Find the second largest element in a single pass. Time O(n). Space O(1).

#include <climits>

int secondLargest(int arr[], int n) {
    if (n < 2) return -1;
    int first = INT_MIN, second = INT_MIN;

    for (int i = 0; i < n; i++) {
        if (arr[i] > first) {
            second = first;       // old max becomes 2nd
            first  = arr[i];      // update max
        } else if (arr[i] > second && arr[i] != first) {
            second = arr[i];      // new 2nd (not a duplicate of max)
        }
    }
    return (second == INT_MIN) ? -1 : second;
}
// Trace [3,1,4,1,5,9,2,6]:
// After scan: first=9, second=6 ✓
// arr[i]!=first prevents duplicate max counting as 2nd

Q15 — Remove Duplicates from a Sorted Array In-Place

Problem: Given a sorted array, remove all duplicates in-place. Return the new size. Time O(n). Space O(1).

int removeDuplicates(int arr[], int n) {
    if (n == 0) return 0;
    int writeIdx = 1;   // arr[0] is always kept

    for (int readIdx = 1; readIdx < n; readIdx++) {
        if (arr[readIdx] != arr[readIdx - 1]) {  // new unique value?
            arr[writeIdx] = arr[readIdx];         // write it
            writeIdx++;
        }
        // duplicate: advance readIdx only, writeIdx stays
    }
    return writeIdx;  // new logical size
}
// Trace [1,1,2,3,3,3,4]:
// r=1: 1==1 → skip | r=2: 2≠1 → arr[1]=2,w=2
// r=3: 3≠2 → arr[2]=3,w=3 | r=4,5: 3==3 → skip×2
// r=6: 4≠3 → arr[3]=4,w=4 → return 4 ✓

Q19 — Two Sum on Sorted Array (Two-Pointer)

Problem: Given a sorted array and a target sum, find two elements that add up to the target. Time O(n). Space O(1).

bool twoSum(int arr[], int n, int target) {
    int left = 0, right = n - 1;
    while (left < right) {
        int sum = arr[left] + arr[right];
        if      (sum == target) {
            cout << arr[left] << " + " << arr[right] << " = " << target << "\n";
            return true;
        }
        else if (sum < target)  left++;   // need larger sum
        else                    right--;  // need smaller sum
    }
    return false;
}
// On sorted [1,2,4,5,7,9], target=11:
// l=0,r=5: 1+9=10<11 → l++
// l=1,r=5: 2+9=11=target → output: 2 + 9 = 11 ✓

Short Answer Questions

Q16. A function receives int arr[], int n as parameters. Why can it NOT use sizeof(arr)/sizeof(arr[0]) to find the array size?

Answer: When passed to a function, an array decays to a pointer to its first element (int*). Inside the function, sizeof(arr) returns the pointer size (8 bytes on 64-bit), NOT the total array size. So sizeof(arr)/sizeof(arr[0]) = 8/4 = 2 regardless of actual array size. Solution: always pass the size as a separate parameter int n.

Q17. What is the difference between stable and unstable sorting? Give an example of each.

Stable sort: equal elements maintain their original relative order after sorting.
Example: Bubble Sort and Insertion Sort — they only swap when arr[j] > arr[j+1] (strict greater-than), so equal elements are never swapped and retain their original sequence.

Unstable sort: equal elements may be reordered.
Example: Selection Sort — finding the minimum in the unsorted portion and swapping it forward can place a later occurrence of an equal value before an earlier one.

When stability matters: when sorting objects by one field while preserving the order from a previous sort on a different field (e.g., sorting students by grade then by name).

Q18. You have 10,000 elements. Compare: (a) 5,000 linear searches vs (b) sort once + 5,000 binary searches. Which is faster?

(a) 5,000 linear searches:
5,000 × O(n) = 5,000 × 10,000 = 50,000,000 operations

(b) Sort once + 5,000 binary searches:
O(n log n) sort ≈ 10,000 × 14 = 140,000 ops
5,000 × O(log n) = 5,000 × 14 = 70,000 ops
Total ≈ 210,000 operations

Option (b) is ~238× faster! General rule: if you search k times and k > log(n), sort once then use binary search every time.

P1 — Build a Linked List & Display It

Problem: Write a complete program that builds a Linked List with values {10, 20, 30, 40} inserted at the END, then displays it as: 10 → 20 → 30 → 40 → NULL

#include <iostream>
using namespace std;

struct Node {
    int   data;
    Node* next;
};

// ── Insert at END ─────────────────────────────────
void insertEnd(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;  n->next = NULL;
    if (head == NULL) { head = n; return; }
    Node* t = head;
    while (t->next != NULL) t = t->next;  // traverse to last node
    t->next = n;
}   // Time: O(n) — must reach the tail first

// ── Display all nodes ─────────────────────────────
void display(Node* head) {
    Node* t = head;
    while (t != NULL) {
        cout << t->data << " → ";
        t = t->next;
    }
    cout << "NULL" << endl;
}

int main() {
    Node* head = NULL;
    insertEnd(head, 10);
    insertEnd(head, 20);
    insertEnd(head, 30);
    insertEnd(head, 40);
    display(head);
    // Output: 10 → 20 → 30 → 40 → NULL
    return 0;
}
List state after each insertEnd:
insertEnd(10) → HEAD → [10|NULL]
insertEnd(20) → HEAD → [10] → [20|NULL]
insertEnd(30) → HEAD → [10] → [20] → [30|NULL]
insertEnd(40) → HEAD → [10] → [20] → [30] → [40|NULL]

P2 — Insert at the Beginning

Problem: Write insertBegin() to insert a new node at the front of the list. Starting from {10, 20, 30}, insert 5 at the beginning → 5 → 10 → 20 → 30 → NULL.

// ── Insert at BEGINNING ─────────────────────────
void insertBegin(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;
    n->next = head;   // new node points to current first node
    head    = n;      // head now points to the new node
}   // Time: O(1) — no traversal needed!

// Before: HEAD → [10] → [20] → [30] → NULL
// insertBegin(head, 5);
// After:  HEAD → [5] → [10] → [20] → [30] → NULL

// ── Step-by-step trace ────────────────────────────
// 1. Create new node n: data=5, next=NULL
// 2. n->next = head  (n now points to old [10])
// 3. head = n        (head now points to new [5])
Key insight — O(1) vs O(n): Inserting at the beginning is always O(1) because we only update two pointers. Inserting at the end is O(n) because we must traverse to the last node first. If you keep a separate tail pointer, end insertion also becomes O(1).

P3 — Insert After a Given Position

Problem: Insert a new node with value val after the node at position pos (1-indexed). List: 10 → 20 → 30 → NULL. Insert 25 after position 2 → 10 → 20 → 25 → 30 → NULL.

// ── Insert after position pos (1-indexed) ─────────
void insertAfter(Node* head, int pos, int val) {
    Node* t = head;
    for (int i = 1; i < pos && t != NULL; i++)
        t = t->next;                     // traverse to node at pos

    if (t == NULL) { cout << "Position out of range\n"; return; }

    Node* n  = new Node();
    n->data  = val;
    n->next  = t->next;   // new node points to what was after t
    t->next  = n;          // t now points to new node
}   // Time: O(pos) — traverse up to pos steps

// Trace: insertAfter(head, 2, 25) on [10→20→30]
// i=1: t moves from [10] to [20]  (i reaches pos=2, stop)
// n->next = t->next = [30]         (new node → 30)
// t->next = n                      (20 → new node)
// Result: 10 → 20 → 25 → 30 → NULL ✓

P4 — Delete the First Node

Problem: Write deleteBegin() to remove the first node. List: 10 → 20 → 30 → NULL → after delete → 20 → 30 → NULL. Don't forget to free memory.

// ── Delete at BEGINNING ─────────────────────────
void deleteBegin(Node*& head) {
    if (head == NULL) { cout << "List is empty!\n"; return; }
    Node* temp = head;       // save pointer to old head
    head = head->next;       // move head to second node
    delete temp;             // free old head memory
}   // Time: O(1)

// Before: HEAD→[10]→[20]→[30]→NULL
// Step 1: temp = head  (temp points to [10])
// Step 2: head = head->next  (head now points to [20])
// Step 3: delete temp  (frees [10] from heap)
// After:  HEAD→[20]→[30]→NULL ✓

P5 — Delete the Last Node

Problem: Write deleteEnd() to remove the last node. List: 10 → 20 → 30 → NULL → after delete → 10 → 20 → NULL. Handle both single-node and multi-node lists.

// ── Delete at END ───────────────────────────────
void deleteEnd(Node*& head) {
    if (!head) return;                    // empty list

    // Special case: only ONE node in list
    if (!head->next) {
        delete head;
        head = NULL;
        return;
    }

    // General case: traverse to second-to-last node
    Node* prev = NULL;
    Node* curr = head;
    while (curr->next != NULL) {
        prev = curr;
        curr = curr->next;    // curr stops at last node
    }
    prev->next = NULL;   // cut the link to last node
    delete curr;          // free last node
}   // Time: O(n)

// Trace on [10→20→30]:
// Loop: prev=[10], curr=[20] → prev=[20], curr=[30] → curr->next=NULL, stop
// prev->next = NULL  (cuts 20's link to 30)
// delete curr        (frees [30])
// Result: 10 → 20 → NULL ✓

P6 — Delete a Node at a Given Position

Problem: Delete the node at position pos (1-indexed). List: 10 → 20 → 30 → 40 → NULL. Delete at position 3 → 10 → 20 → 40 → NULL.

// ── Delete at position pos (1-indexed) ────────────
void deleteAt(Node*& head, int pos) {
    if (!head) return;

    if (pos == 1) { deleteBegin(head); return; }  // O(1) shortcut

    Node* prev = NULL;
    Node* curr = head;
    for (int i = 1; i < pos && curr != NULL; i++) {
        prev = curr;
        curr = curr->next;
    }

    if (!curr) { cout << "Position out of range\n"; return; }

    prev->next = curr->next;   // bypass the node to delete
    delete curr;               // free memory
}   // Time: O(pos)

// Trace: deleteAt(head, 3) on [10→20→30→40]
// i=1: prev=[10], curr=[20]
// i=2: prev=[20], curr=[30]  ← stop, pos=3 reached
// prev->next = curr->next = [40]  (20 now points to 40)
// delete curr   (frees [30])
// Result: 10 → 20 → 40 → NULL ✓

P9 — Complete Program: Insert, Delete & Display

Problem: Build a complete Linked List program that inserts at end (10, 5, 30), displays, deletes first node, displays again, then deletes last node and displays the final result. (This is the exact example from the lab guide.)

#include <iostream>
using namespace std;

struct Node { int data; Node* next; };

void insertEnd(Node*& head, int val) {
    Node* n = new Node(); n->data = val; n->next = NULL;
    if (!head) { head = n; return; }
    Node* t = head;
    while (t->next) t = t->next;
    t->next = n;
}

void deleteBegin(Node*& head) {
    if (!head) return;
    Node* tmp = head; head = head->next; delete tmp;
}

void deleteEnd(Node*& head) {
    if (!head || !head->next) { delete head; head = NULL; return; }
    Node* prev = NULL, *curr = head;
    while (curr->next) { prev = curr; curr = curr->next; }
    prev->next = NULL;  delete curr;
}

void display(Node* head) {
    for (Node* t = head; t; t = t->next) cout << t->data << " → ";
    cout << "NULL\n";
}

int main() {
    Node* head = NULL;
    insertEnd(head, 10);
    insertEnd(head, 5);
    insertEnd(head, 30);

    display(head);         // 10 → 5 → 30 → NULL
    deleteBegin(head);
    display(head);         // 5 → 30 → NULL
    deleteEnd(head);
    display(head);         // 5 → NULL
    return 0;
}

Writing Questions — Short Answer

WQ1 — What is the difference between a node's data and next fields?

Answer: data stores the actual value the node holds (e.g. an integer, string). next is a pointer that stores the memory address of the following node in the list. When next == NULL, the node is the last one in the list. Together they form the fundamental building block: struct Node { int data; Node* next; };

WQ2 — Why do we pass head as Node*& (reference to pointer) in insert/delete functions?

Answer: Because insertions at the beginning and deletions change head itself — they make head point to a different node. If we passed head by value (Node*), only a local copy would change and the original head in main() would remain unchanged. Passing by reference-to-pointer (Node*&) lets the function modify the caller's actual head variable.

WQ3 — What happens if you forget to call delete on a removed node?

Answer: The node's memory on the heap is never returned to the OS. This is called a memory leak. The pointer to the node is lost (no variable holds its address anymore), so the memory becomes permanently inaccessible for the lifetime of the program. In long-running programs, repeated leaks exhaust available memory and cause crashes.

P1 — Bracket Matching

Problem: Write a function that checks if a string has balanced brackets: ( ), [ ], { }. Return true if all brackets are correctly matched and nested.

#include <iostream>
#include <stack>
#include <string>
using namespace std;

bool isBalanced(string s) {
    stack<char> st;
    for (char c : s) {
        // Opening brackets → push onto stack
        if (c == '(' || c == '[' || c == '{')
            st.push(c);
        // Closing bracket → check against top
        else if (c == ')' || c == ']' || c == '}') {
            if (st.empty()) return false;
            char top = st.top(); st.pop();
            if ((c == ')' && top != '(') ||
                (c == ']' && top != '[') ||
                (c == '}' && top != '{'))
                return false;
        }
    }
    return st.empty();  // balanced only if no unmatched openers remain
}

int main() {
    cout << isBalanced("({[]})")  << "\n";  // 1 (true)
    cout << isBalanced("([)]")    << "\n";  // 0 (false — cross-nested)
    cout << isBalanced("{{}}")     << "\n";  // 1 (true)
    cout << isBalanced("(((")      << "\n";  // 0 (false — unclosed)
    return 0;
}
Algorithm: Use a Stack as a memory of unmatched opening brackets. Each closing bracket must match the most recent opening bracket (top of stack). If the stack is non-empty at the end, there are unclosed openers.
Time: O(n) — single pass  ·  Space: O(n) — worst case all openers.

P3 — Reverse a String Using a Stack

Problem: Write a function that reverses a string using a Stack. Do not use the built-in reverse() function.

#include <iostream>
#include <stack>
#include <string>
using namespace std;

string reverseString(string s) {
    stack<char> st;

    // Push every character onto the stack
    for (char c : s)
        st.push(c);

    // Pop characters out — LIFO reverses the order
    string result = "";
    while (!st.empty()) {
        result += st.top();
        st.pop();
    }
    return result;
}

int main() {
    cout << reverseString("HELLO")    << "\n";  // OLLEH
    cout << reverseString("CS211")    << "\n";  // 112SC
    cout << reverseString("abcde")    << "\n";  // edcba
    return 0;
}
Why this works: Pushing all characters then popping them reverses the order because of LIFO. The last character pushed (rightmost) is the first to be popped. Time: O(n) · Space: O(n).

P4 — Count Elements in a Stack (Non-Destructive)

Problem: Count the number of elements currently in a Stack without destroying its contents.

#include <iostream>
using namespace std;

const int MAX = 100;

struct Stack {
    int data[MAX];
    int top = -1;
    void push(int v) { if(top < MAX-1) data[++top]=v; }
    int  pop()       { return top==-1 ? -1 : data[top--]; }
    bool isEmpty()   { return top == -1; }
};

// Pass Stack by VALUE — a copy is made, original is unchanged
int countStack(Stack s) {
    int count = 0;
    while (!s.isEmpty()) {
        s.pop();   // pops from the COPY only
        count++;
    }
    return count;
}

int main() {
    Stack s;
    s.push(10); s.push(20); s.push(30); s.push(40);

    cout << "Count: " << countStack(s) << "\n";  // 4
    cout << "Top still: " << s.data[s.top]    << "\n";  // 40 — original intact
    return 0;
}
Key insight: Pass the Stack by value (Stack s, not Stack& s). C++ makes a full copy of the struct, so popping from inside the function does not affect the original stack.

Advanced Challenge — Queue Using Two Stacks

Problem: Implement a Queue using only two Stacks. Achieve amortized O(1) for both enqueue and dequeue.

#include <iostream>
#include <stack>
using namespace std;

class QueueUsingStacks {
private:
    stack<int> inbox;   // receives all enqueue calls
    stack<int> outbox;  // supplies all dequeue calls

public:
    // ENQUEUE — always push to inbox (O(1))
    void enqueue(int x) {
        inbox.push(x);
    }

    // DEQUEUE — transfer inbox → outbox only when outbox is empty
    int dequeue() {
        if (outbox.empty()) {
            while (!inbox.empty()) {
                outbox.push(inbox.top());   // reversal: LIFO → FIFO
                inbox.pop();
            }
        }
        if (outbox.empty()) { cout << "Queue empty!\n"; return -1; }
        int val = outbox.top();
        outbox.pop();
        return val;
    }

    bool isEmpty() { return inbox.empty() && outbox.empty(); }
};

int main() {
    QueueUsingStacks q;
    q.enqueue(1); q.enqueue(2); q.enqueue(3);
    cout << q.dequeue() << "\n";  // 1 (FIFO order)
    q.enqueue(4);
    cout << q.dequeue() << "\n";  // 2
    cout << q.dequeue() << "\n";  // 3
    cout << q.dequeue() << "\n";  // 4
    return 0;
}

Key insight: Transferring all elements from inbox → outbox reverses their order (LIFO → LIFO = FIFO). The transfer only happens when outbox is empty, so each element is moved at most once → amortized O(1) per dequeue.

Challenge 1 — Count All Nodes in a BST

Problem: Write a recursive function countNodes(Node* root) that returns the total number of nodes in a BST.

#include <iostream>
using namespace std;

struct Node {
    int   data;
    Node* left;
    Node* right;
    Node(int v) : data(v), left(nullptr), right(nullptr) {}
};

int countNodes(Node* root) {
    if (root == nullptr) return 0;   // base case: empty tree
    return 1 + countNodes(root->left)
             + countNodes(root->right);
    // 1 = this node + all in left subtree + all in right subtree
}
// Time: O(n) — visits every node once
// Space: O(h) — call stack depth equals tree height

Node* insert(Node* root, int val) {
    if (!root) return new Node(val);
    if (val < root->data) root->left  = insert(root->left,  val);
    else                  root->right = insert(root->right, val);
    return root;
}

int main() {
    Node* root = nullptr;
    for (int v : {50, 30, 70, 20, 40, 60, 80})
        root = insert(root, v);
    cout << "Node count: " << countNodes(root) << "\n";  // 7
    return 0;
}

Challenge 2 — Find the Height of a BST

Problem: Write a recursive function height(Node* root) that returns the height of a BST. Height = number of edges on the longest path from root to a leaf.

int height(Node* root) {
    if (!root) return -1;   // empty tree: height = -1 (edge convention)
    int lh = height(root->left);
    int rh = height(root->right);
    return 1 + max(lh, rh);   // 1 edge to child + deeper subtree
}
// BST {50,30,70,20,40,60,80} → height = 2
// Time: O(n)  Space: O(h)

Trace on BST {50,30,70,20,40,60,80}:

height(20)=−1+1=0 (leaf)  ·  height(40)=0  ·  height(30)=1+max(0,0)=1

height(60)=0  ·  height(80)=0  ·  height(70)=1

height(50)=1+max(1,1)=2 ✓

Challenge 3 — findMin() and findMax()

Problem: Implement findMin(Node*) and findMax(Node*) for a BST without extra data structures.

// BST property: minimum is always the leftmost node
Node* findMin(Node* root) {
    if (!root) return nullptr;
    while (root->left != nullptr)
        root = root->left;   // keep going left
    return root;
}

// BST property: maximum is always the rightmost node
Node* findMax(Node* root) {
    if (!root) return nullptr;
    while (root->right != nullptr)
        root = root->right;  // keep going right
    return root;
}

// For BST {50,30,70,20,40,60,80}:
// findMin → 20 (leftmost)  findMax → 80 (rightmost)

Challenge 4 — isBST() Validation

Problem: Write a C++ function isBST(Node* root) that returns true if a binary tree satisfies the BST property.

#include <climits>

// Pass valid range [minVal, maxVal] down the recursion
bool isBSTHelper(Node* root, int minVal, int maxVal) {
    if (!root) return true;
    if (root->data <= minVal || root->data >= maxVal)
        return false;
    return isBSTHelper(root->left,  minVal,      root->data) &&
           isBSTHelper(root->right, root->data,  maxVal);
}

bool isBST(Node* root) {
    return isBSTHelper(root, INT_MIN, INT_MAX);
}
// Time: O(n)  Space: O(h)  — passes min/max constraints down the tree
Why we need range constraints: Checking only that left < root < right at each node is NOT enough. Consider root=10, left=5, left.right=15 — 15>10 but was placed as a descendant of the left subtree (where everything must be <10). The range-checking approach catches this.

Q1. What is the in-order predecessor of node 27 in the BST {39, 27, 45, 18, 29, 40, 54}?

Answer: 18

Build the BST from {39,27,45,18,29,40,54}. In-order sequence: 18 → 27 → 29 → 39 → 40 → 45 → 54. The in-order predecessor of 27 is the node that comes immediately before it in sorted order → 18 (the rightmost node of 27's left subtree).

Q2. Prove that in-order traversal of a BST with n distinct integers produces them in O(n) sorted order.

Answer: Correctness: By the BST property, for every node N: all values in N's left subtree < N.data < all values in N's right subtree. In-order visits Left, then Root, then Right — so each node's left subtree values are printed before it, and right subtree values after it. Applied recursively, this yields ascending order. Time O(n): Each of the n nodes is visited exactly once — the traversal function is called once per node, with O(1) work per call (no additional comparison), giving O(n) total.
