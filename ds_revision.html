Question 1 — Struct & Dynamic Array

Lab 4 · Lab 5
Part A
Store information of 12 Students using Struct. Each record: name, registration number, email, mobile number, GPA.
✅ Define a Student struct, declare an array of 12, loop to fill and display:
Question1_partA.cpp C++ 
// Part (a): Store Information of 12 Students Using Struct
#include <iostream>
#include <string>
using namespace std;

struct Student {
    string name;
    string regNumber;
    string email;
    string mobile;
    float  gpa;
};

int main() {
    const int NUM = 12;
    Student students[NUM];

    for (int i = 0; i < NUM; i++) {
        cout << "\nEnter details for Student " << i + 1 << ":\n";
        cout << "Name           : "; getline(cin, students[i].name);
        cout << "Reg. Number    : "; getline(cin, students[i].regNumber);
        cout << "Email          : "; getline(cin, students[i].email);
        cout << "Mobile Number  : "; getline(cin, students[i].mobile);
        cout << "GPA            : "; cin >> students[i].gpa;
        cin.ignore();   // flush newline for next getline
    }

    cout << "\n====== Student Records ======\n";
    for (int i = 0; i < NUM; i++) {
        cout << "\nStudent " << i + 1 << ":\n";
        cout << "  Name    : " << students[i].name      << "\n";
        cout << "  Reg No  : " << students[i].regNumber << "\n";
        cout << "  Email   : " << students[i].email     << "\n";
        cout << "  Mobile  : " << students[i].mobile    << "\n";
        cout << "  GPA     : " << students[i].gpa       << "\n";
    }
    return 0;
}
Key point: After reading a float with cin >>, always call cin.ignore() before the next getline() — otherwise the leftover newline gets consumed as an empty string.
Part B
Create an array whose size is given at run-time, read & store its data, print contents, then remove the whole array from memory.
Question1_partB.cpp C++ 
// Part (b): Dynamic Array with Run-Time Size
#include <iostream>
using namespace std;

int main() {
    int n;
    cout << "Enter array size: ";
    cin  >> n;

    int* arr = new int[n];   // heap allocation — size decided at run-time

    cout << "Enter " << n << " integers:\n";
    for (int i = 0; i < n; i++) {
        cout << "arr[" << i << "] = ";
        cin  >> arr[i];
    }

    cout << "\nArray contents: ";
    for (int i = 0; i < n; i++)
        cout << arr[i] << " ";
    cout << endl;

    delete[] arr;     // release heap memory — MANDATORY
    arr = nullptr;    // good practice: avoid dangling pointer
    cout << "Memory released successfully.\n";
    return 0;
}
Memory rule: Every new must be paired with delete, and every new[] with delete[]. Forgetting causes a memory leak.

Question 2 — Sorting & Searching

Lab 3
Parts A · B · C
Read 20 integers, sort into a separate array using Bubble Sort AND Selection Sort, then search using Linear Search and Binary Search.
Question2.cpp C++ 
#include <iostream>
using namespace std;

const int SIZE = 20;

// ─── Bubble Sort: compare adjacent pairs, bubble largest to end ───
void bubbleSort(int src[], int dst[], int n) {
    for (int i = 0; i < n; i++) dst[i] = src[i]; // copy first
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (dst[j] > dst[j + 1])
                swap(dst[j], dst[j + 1]);
}   // Time: O(n²)  Space: O(1)

// ─── Selection Sort: find minimum, place at front each pass ───
void selectionSort(int src[], int dst[], int n) {
    for (int i = 0; i < n; i++) dst[i] = src[i];
    for (int i = 0; i < n - 1; i++) {
        int minIdx = i;
        for (int j = i + 1; j < n; j++)
            if (dst[j] < dst[minIdx]) minIdx = j;
        swap(dst[i], dst[minIdx]);
    }
}   // Time: O(n²)  Space: O(1)

// ─── Linear Search: check every element ───
int linearSearch(int arr[], int n, int key) {
    for (int i = 0; i < n; i++)
        if (arr[i] == key) return i;
    return -1;
}   // Time: O(n) — works on unsorted arrays

// ─── Binary Search: halve search space each step ───
int binarySearch(int arr[], int n, int key) {
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2; // avoids overflow
        if      (arr[mid] == key) return mid;
        else if (arr[mid]  < key) low  = mid + 1;
        else                      high = mid - 1;
    }
    return -1;
}   // Time: O(log n) — REQUIRES sorted array

int main() {
    int original[SIZE], bubbled[SIZE], selected[SIZE];

    cout << "Enter 20 integers:\n";
    for (int i = 0; i < SIZE; i++) cin >> original[i];

    bubbleSort   (original, bubbled,  SIZE);
    selectionSort(original, selected, SIZE);

    int key;
    cout << "Enter search key: "; cin >> key;

    int li = linearSearch(bubbled, SIZE, key);
    int bi = binarySearch (bubbled, SIZE, key);

    cout << "Linear: " << (li != -1 ? "Found at index " + to_string(li) : "Not found") << "\n";
    cout << "Binary: " << (bi != -1 ? "Found at index " + to_string(bi) : "Not found") << "\n";
    return 0;
}
Time complexity analysis:
Bubble Sort: O(n²) — worst & average. Best case O(n) with an optimised version.
Selection Sort: always O(n²) — always scans the remaining unsorted part.
Linear Search: O(n). Binary Search: O(log n) — requires a sorted array.

Question 3 — Recursive Product

Lab 4 · Recursion
Task
Write a C++ recursive function to find the product of two integer numbers.
The idea: a × b = a + a × (b−1), with base case b == 0 → 0.
Question3.cpp C++ 
#include <iostream>
using namespace std;

// Recursive product: a * b = a + product(a, b-1)
int product(int a, int b) {
    if (b == 0) return 0;          // BASE CASE
    return a + product(a, b - 1);  // RECURSIVE CALL
}

int main() {
    int x, y;
    cout << "Enter two integers: ";
    cin  >> x >> y;

    // Handle negative numbers using XOR for sign detection
    bool negative = (x < 0) ^ (y < 0);
    int result = product(abs(x), abs(y));
    if (negative) result = -result;

    cout << x << " x " << y << " = " << result << endl;
    return 0;
}
Call stack trace for product(3, 3):
product(3,3) → 3 + product(3,2)
product(3,2) → 3 + product(3,1)
product(3,1) → 3 + product(3,0)
product(3,0) → 0 (base case)
Unwind: 3+0=3 → 3+3=6 → 3+6=9 ✓

Question 4 — Fibonacci & Prime Numbers

Lab 3 · Lab 4
Part A
Print the first 50 Fibonacci numbers. Series starts: 0, 1, 1, 2, 3, 5, 8, …
Question4_partA.cpp C++ 
#include <iostream>
using namespace std;

void printFibonacci() {
    long long a = 0, b = 1;
    cout << "First 50 Fibonacci Numbers:\n";
    for (int i = 1; i <= 50; i++) {
        cout << "F(" << i << ") = " << a << "\n";
        long long next = a + b;
        a = b;
        b = next;
    }
}

int main() { printFibonacci(); return 0; }
Why long long? The 50th Fibonacci number is 12,586,269,025 — larger than int (max ~2.1 billion). Use long long to avoid overflow.
Part B
Display prime numbers between two given integers. Analyse time complexity.
Question4_partB.cpp C++ 
#include <iostream>
#include <cmath>
using namespace std;

// Check if n is prime — only test divisors up to √n
bool isPrime(int n) {
    if (n < 2) return false;
    if (n == 2) return true;
    if (n % 2 == 0) return false;
    for (int i = 3; i <= sqrt((double)n); i += 2)
        if (n % i == 0) return false;
    return true;
}

int main() {
    int low, high;
    cout << "Enter lower bound: "; cin >> low;
    cout << "Enter upper bound: "; cin >> high;

    cout << "\nPrimes between " << low << " and " << high << ":\n";
    for (int n = low; n <= high; n++)
        if (isPrime(n)) cout << n << " ";
    cout << endl;
    return 0;
}
Time complexity: For each number n we check divisors up to √n → O(√n) per number. For the range [low, high] with m = high − low numbers → total O(m × √high).

Space complexity: O(1) — only scalar variables used, no extra arrays.

// ── Static array ──────────────────────────────
int nums[5] = {10, 20, 30, 40, 50};

nums[0]  = 99;                  // modify element
cout << nums[2];               // access O(1)

int len = sizeof(nums) / sizeof(nums[0]);  // size = 5

for (int i = 0; i < len; i++)   cout << nums[i] << " ";
for (int x : nums)              cout << x << " ";  // range-for

// ── Dynamic array (heap) ──────────────────────
int n = 5;
int* arr = new int[n];
arr[0] = 10;   arr[1] = 20;
delete[] arr;   arr = nullptr;   // MUST free!

// ── Sorting / searching (algorithm header) ────
#include <algorithm>
sort(nums, nums + 5);                         // ascending
int mx = *max_element(nums, nums + 5);       // max value
int mn = *min_element(nums, nums + 5);       // min value

// ── 2D array ─────────────────────────────────
int mat[2][3] = {{1,2,3},{4,5,6}};
cout << mat[1][2];   // → 6

Search O(n)
Insert/Delete O(n)

linked_list.cppC++
struct Node {
    int   data;
    Node* next;
};

// ── Insert at beginning ────────────────────────
void insertBegin(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;
    n->next = head;  // new node points to old head
    head    = n;     // head now points to new node
}   // O(1)

// ── Insert at end ──────────────────────────────
void insertEnd(Node*& head, int val) {
    Node* n = new Node(); n->data = val; n->next = nullptr;
    if (!head) { head = n; return; }
    Node* t = head;
    while (t->next) t = t->next; // traverse to last
    t->next = n;
}   // O(n)

// ── Delete at beginning ────────────────────────
void deleteBegin(Node*& head) {
    if (!head) return;
    Node* tmp = head;
    head = head->next;
    delete tmp;
}   // O(1)

// ── Delete at end ──────────────────────────────
void deleteEnd(Node*& head) {
    if (!head || !head->next) { delete head; head = nullptr; return; }
    Node* prev = nullptr, *curr = head;
    while (curr->next) { prev = curr; curr = curr->next; }
    prev->next = nullptr;
    delete curr;
}   // O(n)

// ── Display ────────────────────────────────────
void display(Node* head) {
    for (Node* t = head; t; t = t->next)
        cout << t->data << " → ";
    cout << "NULL\n";
}
Access O(n)
Search O(n)
Insert Begin O(1)
Insert End O(n)
Delete Begin O(1)
Space O(n) + ptr

Stack

LIFO · Linear

Last In First Out — like a stack of plates. You can only add or remove from the top.

← TOP (push/pop here)
10
20
30
40 ← TOP
push(50) → 50 becomes new TOP
pop() → removes 40 from TOP
stack_linked.cppC++
struct Node { int data; Node* next; };
Node* top = nullptr;   // global top pointer

// push — insert at beginning (top of stack)
void push(int val) {
    Node* n = new Node(); n->data = val; n->next = top; top = n;
}

// pop — remove from beginning (top of stack)
void pop() {
    if (!top) { cout << "Stack Underflow!\n"; return; }
    Node* tmp = top; top = top->next; delete tmp;
}

// peek — view top without removing
int peek() {
    if (!top) { cout << "Stack Empty!\n"; return -1; }
    return top->data;
}

bool isEmpty() { return top == nullptr; }

// ── or use STL stack (easier) ──────────────────
#include <stack>
std::stack<int> st;
st.push(10);   st.push(20);
cout << st.top();   // 20
st.pop();
cout << st.empty(); // 0 (false)
Use cases: Function call stack, undo/redo, expression evaluation, backtracking (DFS), browser history.
push O(1)
pop O(1)
peek/top O(1)
Search O(n)
Space O(n)

Queue

FIFO · Linear

First In First Out — like a line at a store. You enqueue at the rear and dequeue from the front.

FRONT (dequeue) →
10 ← FRONT
20
30
40 ← REAR
← REAR (enqueue)
queue_linked.cppC++
struct Node { int data; Node* next; };
Node* front = nullptr;
Node* rear  = nullptr;

// enqueue — insert at REAR
void enqueue(int val) {
    Node* n = new Node(); n->data = val; n->next = nullptr;
    if (!rear) { front = rear = n; return; }
    rear->next = n; rear = n;
}   // O(1)

// dequeue — remove from FRONT
void dequeue() {
    if (!front) { cout << "Queue Empty!\n"; return; }
    Node* tmp = front; front = front->next;
    if (!front) rear = nullptr;
    delete tmp;
}   // O(1)

bool isEmpty() { return front == nullptr; }

// ── STL queue ─────────────────────────────────
#include <queue>
std::queue<int> q;
q.push(10); q.push(20); q.push(30);
cout << q.front();   // 10
q.pop();
cout << q.front();   // 20
Use cases: Print queue, CPU scheduling, BFS graph traversal, request handling, simulation.
enqueue O(1)
dequeue O(1)
front/rear O(1)
Search O(n)
Space O(n)

Binary Tree & BST

Non-linear · Hierarchical

A Binary Tree is a hierarchical structure where each node has at most two children (left and right). A Binary Search Tree (BST) adds the rule: left < root < right.

50
/        \
30 70
/   \ /   \
20 40 60 80
BST: left subtree < parent < right subtree
bst.cppC++
struct Node {
    int   data;
    Node* left;
    Node* right;
};

// ── Insert into BST ────────────────────────────
Node* insert(Node* root, int val) {
    if (!root) {
        Node* n = new Node(); n->data = val;
        n->left = n->right = nullptr; return n;
    }
    if      (val < root->data) root->left  = insert(root->left,  val);
    else if (val > root->data) root->right = insert(root->right, val);
    return root;
}

// ── Search in BST ──────────────────────────────
bool search(Node* root, int key) {
    if (!root)             return false;
    if (key == root->data) return true;
    if (key  < root->data) return search(root->left,  key);
                           return search(root->right, key);
}   // O(log n) balanced, O(n) worst case (skewed)

// ── Tree Traversals ────────────────────────────
void inOrder  (Node* r) { if(!r) return; inOrder(r->left); cout<<r->data<<" "; inOrder(r->right); }
void preOrder (Node* r) { if(!r) return; cout<<r->data<<" "; preOrder(r->left); preOrder(r->right); }
void postOrder(Node* r) { if(!r) return; postOrder(r->left); postOrder(r->right); cout<<r->data<<" "; }

// For BST {50,30,70,20,40,60,80}:
// In-order  → 20 30 40 50 60 70 80  (sorted!)
// Pre-order → 50 30 20 40 70 60 80  (root first)
// Post-order→ 20 40 30 60 80 70 50  (root last)
In-order traversal of a BST always gives sorted output.
Height of balanced BST ≈ log₂(n). Height of skewed BST = n (worst case).
Search (balanced) O(log n)
Search (worst) O(n)
Insert (balanced) O(log n)
Traversal O(n)
Space O(n)
