Question 1 — Struct & Dynamic Array

Part (a): Store information of 12 students using a struct. Each record must contain: student name, registration number, email, mobile number, and GPA.

Part (b): Create an array whose size is given at run-time, then read and store its data. Print the array contents, then remove it from memory at run-time.

Part (a) — Student Struct

#include <iostream>
#include <string>
using namespace std;

struct Student {
    string name;
    string regNumber;
    string email;
    string mobile;
    float  gpa;
};

int main() {
    const int NUM = 12;
    Student students[NUM];

    for (int i = 0; i < NUM; i++) {
        cout << "\nStudent " << i + 1 << ":\n";
        cout << "Name           : "; getline(cin, students[i].name);
        cout << "Reg. Number    : "; getline(cin, students[i].regNumber);
        cout << "Email          : "; getline(cin, students[i].email);
        cout << "Mobile Number  : "; getline(cin, students[i].mobile);
        cout << "GPA            : "; cin >> students[i].gpa;
        cin.ignore();   // clear newline before next getline
    }

    cout << "\n====== Student Records ======\n";
    for (int i = 0; i < NUM; i++) {
        cout << "\nStudent " << i + 1 << ":\n";
        cout << "  Name    : " << students[i].name      << "\n";
        cout << "  Reg No  : " << students[i].regNumber << "\n";
        cout << "  Email   : " << students[i].email     << "\n";
        cout << "  Mobile  : " << students[i].mobile    << "\n";
        cout << "  GPA     : " << students[i].gpa       << "\n";
    }
    return 0;
}
Key concepts: struct groups different data types under one name  ·  getline(cin, str) reads the full line including spaces  ·  cin.ignore() clears the newline left by cin >> before a getline.

Part (b) — Dynamic Array (Run-time size)

#include <iostream>
using namespace std;

int main() {
    int n;
    cout << "Enter array size: ";
    cin  >> n;

    int* arr = new int[n];   // allocate on HEAP at run-time

    cout << "Enter " << n << " integers:\n";
    for (int i = 0; i < n; i++) {
        cout << "arr[" << i << "] = ";
        cin  >> arr[i];
    }

    cout << "\nArray contents: ";
    for (int i = 0; i < n; i++)
        cout << arr[i] << " ";
    cout << endl;

    delete[] arr;    // release heap memory — MUST use delete[]
    arr = nullptr;   // avoid dangling pointer
    cout << "Memory released successfully.\n";
    return 0;
}
Stack vs Heap: int arr[5] — stack, fixed, auto-freed  ·  new int[n] — heap, run-time size, must call delete[] manually.

Question 2 — Sorting & Searching

Part (a): Read 20 integers and store them without sorting.

Part (b): Sort into separate arrays using Bubble Sort and Selection Sort.

Part (c): Search using Linear Search and Binary Search.

#include <iostream>
using namespace std;

const int SIZE = 20;

// Bubble Sort — compare adjacent elements, bubble larger ones to the end
void bubbleSort(int src[], int dst[], int n) {
    for (int i = 0; i < n; i++) dst[i] = src[i];           // copy
    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - 1 - i; j++)
            if (dst[j] > dst[j + 1])
                swap(dst[j], dst[j + 1]);
}

// Selection Sort — find minimum, swap to front
void selectionSort(int src[], int dst[], int n) {
    for (int i = 0; i < n; i++) dst[i] = src[i];           // copy
    for (int i = 0; i < n - 1; i++) {
        int minIdx = i;
        for (int j = i + 1; j < n; j++)
            if (dst[j] < dst[minIdx]) minIdx = j;
        swap(dst[i], dst[minIdx]);
    }
}

// Linear Search — O(n) — works on any array
int linearSearch(int arr[], int n, int key) {
    for (int i = 0; i < n; i++)
        if (arr[i] == key) return i;
    return -1;
}

// Binary Search — O(log n) — requires a SORTED array
int binarySearch(int arr[], int n, int key) {
    int low = 0, high = n - 1;
    while (low <= high) {
        int mid = low + (high - low) / 2;
        if      (arr[mid] == key) return mid;
        else if (arr[mid]  < key) low  = mid + 1;
        else                      high = mid - 1;
    }
    return -1;
}

int main() {
    int original[SIZE], bubbled[SIZE], selected[SIZE];

    cout << "Enter 20 integers:\n";
    for (int i = 0; i < SIZE; i++) { cout << "[" << i+1 << "] = "; cin >> original[i]; }

    bubbleSort   (original, bubbled,  SIZE);
    selectionSort(original, selected, SIZE);

    cout << "\nUnsorted      : "; for(int x : original) cout << x << " "; cout << "\n";
    cout << "Bubble Sort   : "; for(int x : bubbled)  cout << x << " "; cout << "\n";
    cout << "Selection Sort: "; for(int x : selected) cout << x << " "; cout << "\n";

    int key;
    cout << "\nEnter search key: "; cin >> key;

    int li = linearSearch(bubbled, SIZE, key);
    int bi = binarySearch (bubbled, SIZE, key);

    cout << "Linear Search : " << (li != -1 ? "Found at index " + to_string(li) : "Not found") << "\n";
    cout << "Binary Search : " << (bi != -1 ? "Found at index " + to_string(bi) : "Not found") << "\n";
    return 0;
}

Question 3 — Recursive Product

Task: Write a recursive C++ function to find the product (multiplication) of two integers.

#include <iostream>
using namespace std;

// Recursive product: a × b = a + a + a ... (b times)
// BASE CASE: b == 0 → return 0
int product(int a, int b) {
    if (b == 0) return 0;          // base case: anything × 0 = 0
    return a + product(a, b - 1); // recursive: a + (a × (b-1))
}

int main() {
    int x, y;
    cout << "Enter two integers: ";
    cin  >> x >> y;

    // Handle negative numbers using XOR sign detection
    bool negative = (x < 0) ^ (y < 0);
    int  result   = product(abs(x), abs(y));
    if (negative) result = -result;

    cout << x << " x " << y << " = " << result << endl;
    return 0;
}

Trace — product(3, 4):

product(3, 4) → 3 + product(3, 3)

product(3, 3) → 3 + product(3, 2)

product(3, 2) → 3 + product(3, 1)

product(3, 1) → 3 + product(3, 0)

product(3, 0) → 0 (BASE CASE)

Unwind: 3+3+3+3+0 = 12 ✓

Question 4 — Fibonacci & Prime Numbers

Part (a): Print the first 50 Fibonacci numbers. Series starts: 0, 1, 1, 2, 3, 5, 8, …

Part (b): Display all prime numbers between two given integers. Analyze time complexity.

#include <iostream>
#include <cmath>
using namespace std;

// Part (a) — First 50 Fibonacci numbers (iterative, avoids recursion overhead)
void partA() {
    long long a = 0, b = 1;
    cout << "First 50 Fibonacci Numbers:\n";
    for (int i = 1; i <= 50; i++) {
        cout << "F(" << i << ") = " << a << "\n";
        long long next = a + b;
        a = b;
        b = next;
    }
}

// Part (b) — Primality test: check divisors up to √n
// Time Complexity: O(√n) per number
bool isPrime(int n) {
    if (n < 2)          return false;
    if (n == 2)         return true;
    if (n % 2 == 0)     return false;
    for (int i = 3; i <= sqrt((double)n); i += 2)
        if (n % i == 0)   return false;
    return true;
}

void partB() {
    int low, high;
    cout << "Enter lower bound: "; cin >> low;
    cout << "Enter upper bound: "; cin >> high;
    cout << "\nPrimes between " << low << " and " << high << ":\n";
    for (int n = low; n <= high; n++)
        if (isPrime(n)) cout << n << " ";
    cout << endl;
}

int main() { partA(); partB(); return 0; }
Time Complexity Analysis: Checking one number for primality = O(√n) because we only test divisors up to the square root. Checking all numbers between low and high = O((high − low) × √high). Space Complexity: O(1) — no extra storage used.

1. Basic Structure of a C++ Program

Question: Explain the basic structure of a C++ program including main() function, namespaces, inputs/outputs, variables, and compiling.

Answer:

A C++ program begins with preprocessor directives such as #include <iostream>, which attach standard library functionality (like input/output streams) before the code is compiled. The line using namespace std; allows the program to use standard library identifiers like cout and cin without writing the std:: prefix each time.

Every C++ program must have a main() function — this is the entry point where execution starts. Inside main(), variables are declared to store data (e.g., int age;, double gpa;, string name;). The cout object (with the << operator) displays output to the screen, while cin (with the >> operator) reads user input from the keyboard.

Once written, the source file (.cpp) is compiled by a compiler such as g++, which translates it into an executable binary. Any syntax errors are reported at this stage.

#include <iostream>    // 1. attach I/O library
using namespace std;   // 2. use std names directly

int main() {            // 3. entry point
    int age;             // 4. declare variable
    cout << "Enter your age: ";   // 5. output
    cin  >> age;                   // 6. input
    cout << "You are " << age << " years old." << endl;
    return 0;            // 7. signal success to OS
}

2. Variables in C++

Question: What are variables and why are they useful in programming? Provide examples of declaring variables of different data types.

Answer:

A variable is a named memory location that stores a value which can be read or changed during program execution. Variables are essential because they allow programs to work with dynamic data — instead of hard-coding values, the program can accept input, perform calculations, and produce different results each run.

C++ is a statically typed language, meaning every variable must be declared with a specific data type before use. Common types include:

  • int — whole numbers (e.g., age, count)
  • double — decimal numbers (e.g., GPA, price)
  • char — single character (e.g., grade 'A')
  • bool — true or false (e.g., passed, isValid)
  • string — text (e.g., name, email)

3. if Statements and Decision Making

Question: Explain how if statements allow decision making and control flow based on conditions. Provide an example.

Answer:

An if statement evaluates a boolean condition. If the condition is true, the associated block of code executes; otherwise it is skipped. This gives programs the ability to branch — taking different paths depending on the data at run-time.

An if can be extended with else if to check additional conditions in sequence, and a final else to handle any case not covered by the previous conditions. Only the first branch whose condition is true will execute.

int score = 75;

if (score >= 90) {
    cout << "Grade: A";       // skipped — 75 < 90
} else if (score >= 75) {
    cout << "Grade: B";       // ← executes (75 >= 75)
} else if (score >= 60) {
    cout << "Grade: C";       // skipped after B matched
} else {
    cout << "Grade: F";       // skipped
}
// Output: Grade: B

4. for Loops vs while Loops

Question: What are for loops and how does their syntax differ from while loops? Give examples showing both loop types.

Answer:

A for loop is used when the number of iterations is known in advance. Its header combines three parts in one line: initialization, condition check, and increment/decrement — making it compact and readable for counted repetition.

A while loop is used when the number of iterations is not known ahead of time — it keeps executing as long as its condition remains true. The initialization happens before the loop, and the update must be written manually inside the body.

// for loop — preferred when count is known
for (int i = 1; i <= 5; i++) {
    cout << i << " ";
}   // Output: 1 2 3 4 5

// while loop — preferred when condition-driven
int i = 1;             // initialization BEFORE loop
while (i <= 5) {
    cout << i << " ";
    i++;               // update INSIDE loop body
}   // Output: 1 2 3 4 5

// do-while — guaranteed to run AT LEAST once
int j = 1;
do {
    cout << j << " ";
    j++;
} while (j <= 5);

    sort(arr2, arr2 + 5);                       // ascending

#include <iostream>
using namespace std;

// ── Step 1: Node structure ──────────────────────────
struct Node {
    int   data;    // the value
    Node* next;    // pointer to the next node
};

// ── Step 2: Insert at END ───────────────────────────
void insertEnd(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;  n->next = NULL;
    if (head == NULL) { head = n; return; }
    Node* t = head;
    while (t->next != NULL) t = t->next;  // traverse to end
    t->next = n;
}

// ── Step 3: Insert at BEGIN ─────────────────────────
void insertBegin(Node*& head, int val) {
    Node* n = new Node();
    n->data = val;
    n->next = head;   // point new node to old head
    head = n;         // update head — O(1)!
}

// ── Step 4: Delete at BEGIN ─────────────────────────
void deleteBegin(Node*& head) {
    if (head == NULL) return;
    Node* temp = head;
    head = head->next;  // advance head
    delete temp;         // free old head
}

// ── Step 5: Delete at END ──────────────────────────
void deleteEnd(Node*& head) {
    if (!head || !head->next) { delete head; head = NULL; return; }
    Node* prev = NULL, *curr = head;
    while (curr->next != NULL) { prev = curr; curr = curr->next; }
    prev->next = NULL;  delete curr;
}

// ── Step 6: Display ────────────────────────────────
void display(Node* head) {
    Node* t = head;
    while (t != NULL) {
        cout << t->data << " → ";
        t = t->next;
    }
    cout << "NULL\n";
}

int main() {
    Node* head = NULL;        // empty list
    insertEnd(head, 10);      // HEAD → 10 → NULL
    insertEnd(head, 20);      // HEAD → 10 → 20 → NULL
    insertEnd(head, 30);      // HEAD → 10 → 20 → 30 → NULL
    display(head);
    deleteBegin(head);        // HEAD → 20 → 30 → NULL
    display(head);
    return 0;
}
Delete at middle: Keep a prev pointer. When you reach the target node, do prev->next = current->next; delete current;. This skips and frees the node in O(n) (traverse) but O(1) relink.

What is it? A Stack follows the LIFO (Last In, First Out) principle — the last element added is the first one removed. Think of a stack of plates: you always add and remove from the top.

#include <iostream>
using namespace std;

const int MAX = 100;

struct Stack {
    int data[MAX];
    int top = -1;    // -1 means empty

    // Push — add element to top
    void push(int val) {
        if (top == MAX - 1) { cout << "Stack Overflow!\n"; return; }
        data[++top] = val;    // increment top, then store
    }

    // Pop — remove and return top element
    int pop() {
        if (top == -1) { cout << "Stack Underflow!\n"; return -1; }
        return data[top--];   // return top, then decrement
    }

    // Peek — view top without removing
    int peek()    { return top == -1 ? -1 : data[top]; }
    bool isEmpty(){ return top == -1; }
};

int main() {
    Stack s;
    s.push(10);   // stack: [10]
    s.push(20);   // stack: [10, 20]
    s.push(30);   // stack: [10, 20, 30]

    cout << "Top: "  << s.peek() << "\n";   // 30
    cout << "Pop: "  << s.pop()  << "\n";   // 30
    cout << "Top: "  << s.peek() << "\n";   // 20
    return 0;
}

Queue

What is it? A Queue follows the FIFO (First In, First Out) principle — the first element added is the first one removed. Think of a checkout line: new people join at the back, and are served from the front.

Search

Real uses: Print queues, CPU scheduling, BFS graph traversal, keyboard input buffer.

#include <iostream>
using namespace std;

const int MAX = 100;

struct Queue {
    int  data[MAX];
    int  front = 0;
    int  back  = -1;
    int  size  = 0;

    // Enqueue — add to back
    void enqueue(int val) {
        if (size == MAX) { cout << "Queue Full!\n"; return; }
        back = (back + 1) % MAX;   // circular wrap
        data[back] = val;
        size++;
    }

    // Dequeue — remove from front
    int dequeue() {
        if (size == 0) { cout << "Queue Empty!\n"; return -1; }
        int val = data[front];
        front = (front + 1) % MAX;  // circular wrap
        size--;
        return val;
    }

    int  peek()    { return size == 0 ? -1 : data[front]; }
    bool isEmpty(){ return size == 0; }
};

int main() {
    Queue q;
    q.enqueue(10);   // queue: [10]
    q.enqueue(20);   // queue: [10, 20]
    q.enqueue(30);   // queue: [10, 20, 30]

    cout << "Front:   " << q.peek()     << "\n";   // 10
    cout << "Dequeue: " << q.dequeue() << "\n";   // 10
    cout << "Front:   " << q.peek()     << "\n";   // 20
    return 0;
}
Stack vs Queue: Both are linear data structures with O(1) add/remove. The key difference is direction — Stack removes from the same end it adds (LIFO), Queue removes from the opposite end (FIFO).

Binary Tree

🌳
Binary Tree / BST
Hierarchical · Each node has at most 2 children

What is it? A tree is a hierarchical (non-linear) data structure. Each element is a node. There is one root at the top. Each node has at most two children: left and right. A node with no children is called a leaf.

Binary Search Tree (BST) rule: left child < parent < right child — this makes searching very fast.

Search (BST)
O(log n)
Insert (BST)
O(log n)
Traversal
O(n)
Worst case
O(n)

Real uses: File systems, expression parsing, database indexes, autocomplete, game AI (decision trees).

#include <iostream>
using namespace std;

// ── Node structure ─────────────────────────────────
struct TreeNode {
    int       data;
    TreeNode* left;
    TreeNode* right;
    TreeNode(int val) : data(val), left(NULL), right(NULL) {}
};

// ── Insert into BST ────────────────────────────────
TreeNode* insert(TreeNode* root, int val) {
    if (root == NULL) return new TreeNode(val);
    if      (val < root->data) root->left  = insert(root->left,  val);
    else if (val > root->data) root->right = insert(root->right, val);
    return root;   // duplicates are ignored
}

// ── Inorder: Left → Root → Right (gives SORTED order) ─
void inorder(TreeNode* root) {
    if (root == NULL) return;
    inorder(root->left);
    cout << root->data << " ";
    inorder(root->right);
}

// ── Preorder: Root → Left → Right ─────────────────
void preorder(TreeNode* root) {
    if (root == NULL) return;
    cout << root->data << " ";
    preorder(root->left);
    preorder(root->right);
}

// ── Postorder: Left → Right → Root ────────────────
void postorder(TreeNode* root) {
    if (root == NULL) return;
    postorder(root->left);
    postorder(root->right);
    cout << root->data << " ";
}

// ── BST Search ─────────────────────────────────────
bool search(TreeNode* root, int key) {
    if (root == NULL)       return false;
    if (root->data == key)   return true;
    if (key < root->data)    return search(root->left,  key);
    else                     return search(root->right, key);
}

int main() {
    TreeNode* root = NULL;
    for (int v : {50, 30, 70, 20, 40, 60, 80})
        root = insert(root, v);

    cout << "Inorder  : "; inorder  (root); cout << "\n";
    cout << "Preorder : "; preorder (root); cout << "\n";
    cout << "Postorder: "; postorder(root); cout << "\n";

    cout << "Search 40: " << (search(root, 40) ? "Found" : "Not found") << "\n";
    cout << "Search 99: " << (search(root, 99) ? "Found" : "Not found") << "\n";
    return 0;
}

How to choose a structure:

  • Need fast index access? → Array
  • Lots of insertions/deletions? → Linked List
  • Undo / function calls / backtracking? → Stack
  • Scheduling / print queue / BFS? → Queue
  • Need sorted data + fast search? → Binary Search Tree
